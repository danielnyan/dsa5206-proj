# PANDA CPU chunked preparation + export addendum
# =================================================
#
# PURPOSE
# -------
# Replace the original unbounded A6/A7 preparation/handoff cells AND the older
# "export after everything is prepared" addendum.
#
# This version prepares and exports PANDA in deterministic bounded chunks. It is
# intended for Kaggle sessions with a time/output-size limit and for eventual
# export of the *entire* eligible cohort over multiple runs.
#
# IMPORTANT FOR EXISTING NOTEBOOKS
# --------------------------------
# Keep A1-A5 unchanged, including the completed inventory in PREP_ROOT.
#
# DO NOT run the old A6:
#
#     prepare_selected(... CAPS ...)
#
# and DO NOT run the old single-ZIP A8 exporter.
#
# Put this block immediately after A5 instead.
#
# The chunker deliberately uses the same final ordering as the original
# `select_tiles(..., caps=None)` path:
#
#     (image_id, y, x, tile_id)
#
# Therefore an interrupted legacy CAPS=None preparation can be salvaged. If
# PREP_ROOT/cache + PREP_ROOT/tile_ledger.jsonl already contain normalized tiles,
# this block reuses them when they belong to the selected chunk.
#
# CHUNK SEMANTICS
# ---------------
# CHUNK_INDEX is zero-based. CHUNK_TILE_COUNT is editable on every *new* chunk.
# The next chunk starts automatically at the previous completed chunk's
# selection_stop recorded in PREP_ROOT/chunk_exports.jsonl.
#
# Example:
#
#     chunk 0: CHUNK_TILE_COUNT=40000 -> eligible rows      0:40000
#     chunk 1: CHUNK_TILE_COUNT=30000 -> eligible rows  40000:70000
#     chunk 2: CHUNK_TILE_COUNT=50000 -> eligible rows  70000:120000
#
# The eligible-row ordering is deterministic, so completed chunks are disjoint
# and their union covers the cohort without requiring equal chunk sizes. When
# resuming an interrupted chunk, keep BOTH CHUNK_INDEX and CHUNK_TILE_COUNT
# unchanged.
#
# A chunk is resumable. Incomplete normalized PNGs and the chunk ledger live
# under PREP_ROOT/chunk_work/chunk_XXXX, so `restore_previous()` carries them
# forward with the ordinary PREVIOUS_CPU_OUTPUT workflow.
#
# Once a chunk ZIP is finalized and SHA-256 verified, this script can delete the
# corresponding raw PNG cache. Small chunk metadata are retained so that the
# next notebook version does not carry all previous image caches forward.
#
# The final ZIP is a transfer shard, not necessarily a standalone training set.
# Extract all desired shards on HPC and merge them with the companion
# PANDA_HPC_merge_chunks.py helper before running production PBS training.
#
# RECOMMENDED FIRST CHUNK FOR THE CURRENT RUN
# -------------------------------------------
# The observed cache was ~6.72 GiB at ~38,100 processed candidates. A 40,000-tile
# shard should therefore be around 7 GiB at the same compression ratio and will
# reuse almost all of the work already done in the interrupted legacy run.
#
# Future users may edit CHUNK_TILE_COUNT. Keep it small enough that BOTH the raw
# chunk cache and the temporary ZIP can coexist in /kaggle/working while the ZIP
# is being finalized.
#
# ---------------------------------------------------------------------------
# A6C.1 — user configuration
# ---------------------------------------------------------------------------

CHUNK_TILE_COUNT = 81_000

# Archive is intentionally OUTSIDE PREP_ROOT. restore_previous() therefore
# restores only the small resumable state, not all old ZIPs.

# ---------------------------------------------------------------------------
# A6C.3 — deterministic contiguous chunk selection from completed inventory
# ---------------------------------------------------------------------------

def _chunk_selection(
    inventory_root,
    chunk_index,
    chunk_tile_count,
    start,
    min_tumour_fraction=0.0,
):
    """
    Return one contiguous shard of the complete eligible cohort.

    Ordering intentionally matches the final ordering of the old
    select_tiles(..., caps=None) implementation:
        image_id, y, x
    with tile_id as a deterministic tie-breaker.
    """
    inventory_root = Path(inventory_root)
    db = sqlite3.connect(inventory_root / 'inventory.sqlite')
    db.row_factory = sqlite3.Row

    clause = eligibility_clause(min_tumour_fraction)

    total = db.execute(
        'SELECT COUNT(*) '
        'FROM tiles t JOIN slides s USING(image_id) '
        'WHERE ' + clause,
        (min_tumour_fraction,),
    ).fetchone()[0]

    start = int(start)

    if start < 0:
        db.close()
        raise ValueError('Chunk start offset must be >= 0')

    if start >= total:
        db.close()
        return [], {
            'chunk_index': int(chunk_index),
            'chunk_tile_count': int(chunk_tile_count),
            'global_eligible': int(total),
            'start': int(start),
            'stop': int(start),
            'selected': 0,
            'is_last_chunk': True,
            'all_chunks_exported': True,
        }

    rows = [
        dict(row)
        for row in db.execute(
            'SELECT '
            't.*, '
            's.split, s.role, s.gleason_score, s.expected_gp, '
            's.mpp_x, s.mpp_y, s.actual_fov_um_x, s.actual_fov_um_y '
            'FROM tiles t JOIN slides s USING(image_id) '
            'WHERE ' + clause + ' '
            'ORDER BY t.image_id, t.y, t.x, t.tile_id '
            'LIMIT ? OFFSET ?',
            (
                min_tumour_fraction,
                int(chunk_tile_count),
                int(start),
            ),
        )
    ]

    db.close()

    stop = start + len(rows)

    selection_sha256 = hashlib.sha256(
        '\n'.join(row['tile_id'] for row in rows).encode('utf-8')
    ).hexdigest()

    meta = {
        'chunk_index': int(chunk_index),
        'chunk_tile_count': int(chunk_tile_count),
        'global_eligible': int(total),
        'start': int(start),
        'stop': int(stop),
        'selected': len(rows),
        'is_last_chunk': stop >= total,
        'all_chunks_exported': False,
        'selection_order': 'image_id,y,x,tile_id',
        'selection_sha256': selection_sha256,
    }

    return rows, meta


def _next_chunk_start(prep_root, chunk_index):
    """
    Resolve the start offset for a new/resumed chunk.

    * chunk 0 always starts at 0.
    * for chunk N>0, start at selection_stop of completed chunk N-1.
    * if chunk N already has chunk_settings.json (resume), use its frozen
      selection_start instead.

    This makes CHUNK_TILE_COUNT editable between completed chunks without
    creating gaps or overlaps.
    """
    prep_root = Path(prep_root)
    chunk_root = prep_root / 'chunk_work' / f'chunk_{chunk_index:04d}'
    frozen = chunk_root / 'chunk_settings.json'

    if frozen.exists():
        settings = json.loads(frozen.read_text())
        if int(settings['chunk_index']) != int(chunk_index):
            raise ValueError('Frozen chunk index mismatch')
        return int(settings['selection_start'])

    if chunk_index == 0:
        return 0

    exports = [
        row
        for row in read_jsonl(prep_root / 'chunk_exports.jsonl')
        if int(row.get('chunk_index', -1)) == int(chunk_index) - 1
    ]

    if not exports:
        raise ValueError(
            f'Cannot start chunk {chunk_index}: completed chunk '
            f'{chunk_index - 1} is not recorded in chunk_exports.jsonl. '
            'Resume/finish the previous chunk first.'
        )

    previous = exports[-1]
    return int(previous['selection_stop'])


# ---------------------------------------------------------------------------
# A6C.4 — HPC manifest conversion
# ---------------------------------------------------------------------------

def _hpc_row(row):
    label = int(row['label'])
    gp = label + 3

    return {
        'dataset': 'panda',
        'split': row['split'],
        'sample_id': f'panda:{row["tile_id"]}',
        'class_name': CLASSES[label],
        'label': label,

        # Existing pure-cohort contract used by the gleason branch.
        # PANDA training eligibility was already decided during inventory.
        'region_role': (
            'pure'
            if row['split'] == 'train'
            else 'heldout_single_gp'
        ),
        'region_primary': gp,
        'region_secondary': gp,

        'source_wsi': row['image_id'],
        'patient_id': '',

        # The normalized PNG is the object passed to model training.
        'file_sha256': row['cache_sha256'],
        'cache_file': row['cache_file'],
        'cache_sha256': row['cache_sha256'],

        # Preserve PANDA/FOV provenance.
        'tile_id': row['tile_id'],
        'image_id': row['image_id'],
        'x': row['x'],
        'y': row['y'],
        'tile_width': row['tile_width'],
        'tile_height': row['tile_height'],
        'mpp_x': row['mpp_x'],
        'mpp_y': row['mpp_y'],
        'actual_fov_um_x': row['actual_fov_um_x'],
        'actual_fov_um_y': row['actual_fov_um_y'],
        'tumour_fraction': row['tumour_fraction'],
        'gleason_score': row.get('gleason_score', ''),
        'expected_gp': row.get('expected_gp', ''),
    }


def _write_hpc_manifest(path, rows):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)

    with path.open('w', newline='') as stream:
        writer = csv.DictWriter(stream, fieldnames=HPC_FIELDS)
        writer.writeheader()
        writer.writerows(rows)


# ---------------------------------------------------------------------------
# A6C.5 — cache lookup / integrity
# ---------------------------------------------------------------------------

def _load_latest_jsonl(path, key):
    result = {}
    for row in read_jsonl(path):
        if key in row:
            result[row[key]] = row
    return result


def _valid_cached(path, expected_sha):
    path = Path(path)
    return (
        path.is_file()
        and expected_sha
        and sha(path) == expected_sha
    )


def _resolve_cache_source(item):
    """
    Resolve the PNG used for final archive creation.

    Prefer chunk-private cache. Fall back to legacy PREP_ROOT/cache for tiles
    salvaged from an interrupted old preparation.
    """
    name = item['cache_file']

    candidate = CHUNK_CACHE / name
    if _valid_cached(candidate, item['cache_sha256']):
        return candidate

    candidate = LEGACY_CACHE / name
    if _valid_cached(candidate, item['cache_sha256']):
        return candidate

    raise FileNotFoundError(
        f'No verified cache file for {item["tile_id"]}: {name}'
    )


# ---------------------------------------------------------------------------
# A6C.6 — resumable preparation of one shard
# ---------------------------------------------------------------------------

def prepare_chunk(
    selected,
    selection_meta,
    reference,
    deadline=None,
):
    CHUNK_ROOT.mkdir(parents=True, exist_ok=True)
    CHUNK_CACHE.mkdir(parents=True, exist_ok=True)
    CHUNK_MANIFESTS.mkdir(parents=True, exist_ok=True)

    done_marker = CHUNK_ROOT / 'chunk_done.json'
    if done_marker.exists() and not ALLOW_REEXPORT_COMPLETED_CHUNK:
        previous = json.loads(done_marker.read_text())
        raise RuntimeError(
            f'{CHUNK_NAME} was already completed. '
            f'Use CHUNK_INDEX={CHUNK_INDEX + 1} for the next shard, '
            f'or set ALLOW_REEXPORT_COMPLETED_CHUNK=True deliberately. '
            f'Previous archive SHA-256: {previous.get("archive_sha256")}'
        )

    inventory_settings = json.loads(
        (PREP_ROOT / 'inventory_settings.json').read_text()
    )

    settings = {
        'pipeline_version': PIPELINE_VERSION,
        'chunk_protocol': 'contiguous-eligible-v1',
        'chunk_index': CHUNK_INDEX,
        'chunk_tile_count': CHUNK_TILE_COUNT,
        'selection_start': selection_meta['start'],
        'selection_stop': selection_meta['stop'],
        'selection_sha256': selection_meta['selection_sha256'],
        'global_eligible': selection_meta['global_eligible'],
        'inventory_settings_sha256': sha(
            PREP_ROOT / 'inventory_settings.json'
        ),
        'reference_sha256': sha(reference),
        'target_fov_um': inventory_settings['target_fov_um'],
        'output_pixels': inventory_settings['output_pixels'],
        'min_tumour_fraction': MIN_TUMOUR_FRACTION,
        'normalization': (
            'direct ~150um source crop -> LANCZOS 350 -> '
            'legacy brightness -> StainTools Macenko -> uint8 PNG'
        ),
    }

    settings_path = CHUNK_ROOT / 'chunk_settings.json'

    if settings_path.exists():
        previous_settings = json.loads(settings_path.read_text())
        if previous_settings != settings:
            raise ValueError(
                f'{CHUNK_NAME} settings changed. '
                'Resume with identical chunk settings or choose a new CHUNK_INDEX.'
            )

    atomic_json(settings_path, settings)

    chunk_ledger_path = CHUNK_ROOT / 'chunk_ledger.jsonl'
    chunk_exclusions_path = CHUNK_ROOT / 'chunk_exclusions.jsonl'

    chunk_previous = _load_latest_jsonl(
        chunk_ledger_path,
        'tile_id',
    )
    chunk_excluded = _load_latest_jsonl(
        chunk_exclusions_path,
        'tile_id',
    )

    legacy_previous = (
        _load_latest_jsonl(LEGACY_LEDGER, 'tile_id')
        if REUSE_LEGACY_CACHE and LEGACY_LEDGER.exists()
        else {}
    )

    legacy_excluded = (
        _load_latest_jsonl(LEGACY_STAIN_EXCLUSIONS, 'tile_id')
        if LEGACY_STAIN_EXCLUSIONS.exists()
        else {}
    )

    normalizers = make_normalizer(reference)

    completed = []
    excluded = []
    current = None
    slide = None
    started = time.monotonic()
    new_cache_bytes = sum(
        p.stat().st_size
        for p in CHUNK_CACHE.glob('*.png')
        if p.is_file()
    )
    stop_reason = None

    try:
        for number, row in enumerate(selected, 1):
            check_deadline(deadline)

            tile_id = row['tile_id']

            # Already known exclusion from this chunk.
            if tile_id in chunk_excluded:
                excluded.append(chunk_excluded[tile_id])
                continue

            # Preserve a stain-normalization exclusion made by the legacy run.
            if tile_id in legacy_excluded:
                event = dict(
                    legacy_excluded[tile_id],
                    chunk_index=CHUNK_INDEX,
                    source='legacy_stain_exclusion',
                )
                append_audit(chunk_exclusions_path, event)
                chunk_excluded[tile_id] = event
                excluded.append(event)
                continue

            # Resume chunk-private completion.
            old = chunk_previous.get(tile_id)
            if old is not None:
                try:
                    _resolve_cache_source(old)
                    completed.append(old)
                    continue
                except FileNotFoundError:
                    # Ledger exists but PNG is unavailable. Rebuild safely.
                    pass

            # Reuse legacy cache produced by the old CAPS=None run.
            legacy = legacy_previous.get(tile_id)
            if legacy is not None:
                legacy_file = LEGACY_CACHE / legacy['cache_file']

                if _valid_cached(
                    legacy_file,
                    legacy.get('cache_sha256'),
                ):
                    item = dict(
                        row,
                        cache_file=legacy['cache_file'],
                        cache_sha256=legacy['cache_sha256'],
                        cache_origin='legacy',
                    )
                    append_audit(chunk_ledger_path, item)
                    chunk_previous[tile_id] = item
                    completed.append(item)

                    if number % 100 == 0:
                        print(
                            f'Chunk {CHUNK_INDEX}: '
                            f'{number}/{len(selected)} candidates; '
                            f'reused legacy cache; '
                            f'new cache {new_cache_bytes / 1024**3:.2f} GiB',
                            flush=True,
                        )

                    continue

            # Guard the amount of NEW chunk-private PNG data. This does not
            # estimate the final ZIP; choose CHUNK_TILE_COUNT conservatively.
            raw_upper = OUTPUT_PIXELS * OUTPUT_PIXELS * 3
            if (
                new_cache_bytes + raw_upper
                > CHUNK_MAX_NEW_CACHE_GIB * 1024**3
            ):
                stop_reason = (
                    'new_chunk_cache_size_guard; '
                    'reduce CHUNK_TILE_COUNT or increase '
                    'CHUNK_MAX_NEW_CACHE_GIB'
                )
                break

            if current != row['image_id']:
                if slide is not None:
                    slide.close()

                slide = openslide.OpenSlide(
                    str(
                        DATA_ROOT
                        / 'train_images'
                        / f'{row["image_id"]}.tiff'
                    )
                )
                current = row['image_id']

            tile = rgb_region(
                slide,
                int(row['x']),
                int(row['y']),
                int(row['tile_width']),
                int(row['tile_height']),
            )

            try:
                pixels = normalize_tile(
                    tile,
                    normalizers,
                    output_pixels=OUTPUT_PIXELS,
                )
            except Exception as error:
                event = {
                    'tile_id': tile_id,
                    'image_id': row['image_id'],
                    'x': row['x'],
                    'y': row['y'],
                    'split': row['split'],
                    'label': row['label'],
                    'chunk_index': CHUNK_INDEX,
                    'error': type(error).__name__,
                    'message': str(error),
                    'source': 'chunk_preparation',
                }

                append_audit(
                    chunk_exclusions_path,
                    event,
                )
                chunk_excluded[tile_id] = event
                excluded.append(event)

                warnings.warn(
                    f'Excluded {tile_id}: '
                    f'{type(error).__name__}: {error}',
                    RuntimeWarning,
                )
                continue

            cached = CHUNK_CACHE / f'{tile_id}.png'
            temporary = cached.with_suffix('.tmp')

            Image.fromarray(pixels).save(
                temporary,
                format='PNG',
            )

            temporary.replace(cached)
            new_cache_bytes += cached.stat().st_size

            item = dict(
                row,
                cache_file=cached.name,
                cache_sha256=sha(cached),
                cache_origin='chunk',
            )

            append_audit(
                chunk_ledger_path,
                item,
            )
            chunk_previous[tile_id] = item
            completed.append(item)

            if number % 100 == 0:
                print(
                    f'Chunk {CHUNK_INDEX}: '
                    f'{number}/{len(selected)} candidates; '
                    f'new cache {new_cache_bytes / 1024**3:.2f} GiB',
                    flush=True,
                )

    except BudgetStop:
        stop_reason = (
            'session_budget; resume this same CHUNK_INDEX '
            'from PREVIOUS_CPU_OUTPUT next run'
        )

    finally:
        if slide is not None:
            slide.close()

    completed_by_id = {
        row['tile_id']: row
        for row in completed
    }
    excluded_ids = {
        row['tile_id']
        for row in excluded
    }

    # Reconstruct completion from ledgers so duplicate-free counts remain correct
    # after a resumed run.
    selected_ids = {
        row['tile_id']
        for row in selected
    }

    ledger_now = _load_latest_jsonl(
        chunk_ledger_path,
        'tile_id',
    )
    exclusions_now = _load_latest_jsonl(
        chunk_exclusions_path,
        'tile_id',
    )

    retained_ids = selected_ids & set(ledger_now)
    excluded_ids = selected_ids & set(exclusions_now)

    processed_ids = retained_ids | excluded_ids
    complete = (
        stop_reason is None
        and processed_ids == selected_ids
    )

    progress = {
        **selection_meta,
        'complete': complete,
        'retained': len(retained_ids),
        'excluded': len(excluded_ids),
        'processed': len(processed_ids),
        'remaining': len(selected_ids - processed_ids),
        'new_cache_bytes': new_cache_bytes,
        'elapsed_seconds_this_run': time.monotonic() - started,
        'stop_reason': stop_reason,
    }

    atomic_json(
        CHUNK_ROOT / 'chunk_progress.json',
        progress,
    )

    if not complete:
        print(json.dumps(progress, indent=2))
        return None, progress

    completed = [
        ledger_now[row['tile_id']]
        for row in selected
        if row['tile_id'] in ledger_now
    ]

    return completed, progress


# ---------------------------------------------------------------------------
# A6C.7 — finalize manifests + transfer ZIP atomically
# ---------------------------------------------------------------------------

def finalize_chunk_archive(
    completed,
    progress,
    reference,
):
    train_rows = [
        _hpc_row(row)
        for row in completed
        if row['split'] == 'train'
    ]
    validation_rows = [
        _hpc_row(row)
        for row in completed
        if row['split'] == 'validation'
    ]

    _write_hpc_manifest(
        CHUNK_MANIFESTS / 'panda_train.csv',
        train_rows,
    )
    _write_hpc_manifest(
        CHUNK_MANIFESTS / 'panda_validation.csv',
        validation_rows,
    )

    counts = {}
    for split, rows in (
        ('train', train_rows),
        ('validation', validation_rows),
    ):
        labels = Counter(int(row['label']) for row in rows)
        counts[split] = {
            CLASSES[k]: int(labels.get(k, 0))
            for k in range(3)
        }

    chunk_meta = {
        'dataset': 'panda',
        'provider': 'radboud',
        'chunk_index': CHUNK_INDEX,
        'chunk_tile_count': CHUNK_TILE_COUNT,
        'selection_start': progress['start'],
        'selection_stop': progress['stop'],
        'global_eligible': progress['global_eligible'],
        'selection_sha256': progress['selection_sha256'],
        'selection_order': progress['selection_order'],
        'is_last_chunk': progress['is_last_chunk'],
        'selected': progress['selected'],
        'retained': len(completed),
        'excluded': progress['excluded'],
        'counts': counts,
        'target_fov_um': TARGET_FOV_UM,
        'output_pixels': OUTPUT_PIXELS,
        'reference_sha256': sha(reference),
        'inventory_settings_sha256': sha(
            PREP_ROOT / 'inventory_settings.json'
        ),
        'training_seed_definition': (
            'PANDA Radboud pure-source proxy: train slides require '
            'published pure 3+3/4+4/5+5 score and matching single-GP '
            'whole-mask pattern; validation uses held-out-slide '
            'single-GP tiles'
        ),
        'patient_independence': (
            'unverified: public PANDA image IDs are not patient identifiers'
        ),
        'preprocessing': (
            'MPP-aware ~150um level-0 crop -> LANCZOS 350 -> '
            'legacy brightness -> StainTools Macenko -> uint8 PNG; '
            'float32 /255 deferred to model training'
        ),
        'note': (
            'This is a transfer shard. Merge all required shards on HPC '
            'before production training.'
        ),
    }

    atomic_json(
        CHUNK_ROOT / 'chunk.json',
        chunk_meta,
    )

    manifests_sha = {
        name: sha(CHUNK_MANIFESTS / name)
        for name in (
            'panda_train.csv',
            'panda_validation.csv',
        )
    }

    atomic_json(
        CHUNK_ROOT / 'manifest_hashes.json',
        manifests_sha,
    )

    handoff = CHUNK_ROOT / 'HPC_HANDOFF.txt'
    handoff.write_text(
        f"""PANDA chunked HPC handoff

Shard:
  {CHUNK_BUNDLE_NAME}

Chunk index:
  {CHUNK_INDEX}

Selection interval in the deterministic eligible cohort:
  [{progress['start']}, {progress['stop']})

Global eligible candidates:
  {progress['global_eligible']}

Retained normalized PNGs:
  {len(completed)}

This archive is a TRANSFER SHARD.
Do not point train_gleason.pbs directly at a collection of shard directories.

On HPC:
  1. Extract every desired panda_150um_chunk_XXXX.zip under one parent.
  2. Run PANDA_HPC_merge_chunks.py on that parent.
  3. Point PREPARED_DIR at the merged output.
  4. Keep DATASET=panda and TRAINING_COHORT=pure.
  5. The NASNetLarge architecture / ImageNet initialization / 14-epoch
     training schedule remain unchanged.

For a full-cohort merge, provide every contiguous chunk from 0000 through the
chunk whose chunk.json has is_last_chunk=true. The merge helper refuses gaps
unless --allow-partial is explicitly supplied.
"""
    )

    # Build to .partial and rename only after ZipFile closes successfully.
    if CHUNK_ARCHIVE_PARTIAL.exists():
        CHUNK_ARCHIVE_PARTIAL.unlink()

    if CHUNK_ARCHIVE.exists():
        CHUNK_ARCHIVE.unlink()

    with zipfile.ZipFile(
        CHUNK_ARCHIVE_PARTIAL,
        mode='w',
        compression=zipfile.ZIP_STORED,
        allowZip64=True,
    ) as zf:
        prefix = CHUNK_BUNDLE_NAME.rstrip('/') + '/'

        for path, arcname in (
            (
                CHUNK_ROOT / 'chunk.json',
                prefix + 'chunk.json',
            ),
            (
                CHUNK_ROOT / 'chunk_settings.json',
                prefix + 'chunk_settings.json',
            ),
            (
                CHUNK_ROOT / 'manifest_hashes.json',
                prefix + 'manifest_hashes.json',
            ),
            (
                CHUNK_MANIFESTS / 'panda_train.csv',
                prefix + 'manifests/panda_train.csv',
            ),
            (
                CHUNK_MANIFESTS / 'panda_validation.csv',
                prefix + 'manifests/panda_validation.csv',
            ),
            (
                handoff,
                prefix + 'HPC_HANDOFF.txt',
            ),
        ):
            zf.write(path, arcname)

        seen = set()

        for number, item in enumerate(completed, 1):
            name = item['cache_file']

            if name in seen:
                raise ValueError(
                    f'Duplicate cache filename inside shard: {name}'
                )
            seen.add(name)

            source = _resolve_cache_source(item)

            # Recheck immediately before archive write.
            if sha(source) != item['cache_sha256']:
                raise ValueError(
                    f'Cache checksum changed before archive: '
                    f'{item["tile_id"]}'
                )

            zf.write(
                source,
                prefix + 'cache/' + name,
            )

            if number % 500 == 0:
                print(
                    f'Archived {number:,}/{len(completed):,} PNGs',
                    flush=True,
                )

    CHUNK_ARCHIVE_PARTIAL.replace(
        CHUNK_ARCHIVE
    )

    archive_sha = sha(
        CHUNK_ARCHIVE
    )

    CHUNK_ARCHIVE_SHA.write_text(
        f'{archive_sha}  {CHUNK_ARCHIVE.name}\n'
    )

    result = {
        **chunk_meta,
        'archive': str(CHUNK_ARCHIVE),
        'archive_sha256': archive_sha,
        'archive_bytes': CHUNK_ARCHIVE.stat().st_size,
        'archive_GiB': CHUNK_ARCHIVE.stat().st_size / 1024**3,
    }

    # Record success BEFORE optional raw-cache cleanup.
    atomic_json(
        CHUNK_ROOT / 'chunk_done.json',
        result,
    )

    append_audit(
        PREP_ROOT / 'chunk_exports.jsonl',
        {
            'chunk_index': CHUNK_INDEX,
            'archive_name': CHUNK_ARCHIVE.name,
            'archive_sha256': archive_sha,
            'selection_start': progress['start'],
            'selection_stop': progress['stop'],
            'global_eligible': progress['global_eligible'],
            'is_last_chunk': progress['is_last_chunk'],
            'retained': len(completed),
            'excluded': progress['excluded'],
        },
    )

    # Only remove raw PNGs after the final archive hash exists.
    if DELETE_RAW_CACHE_AFTER_ARCHIVE:
        if CHUNK_CACHE.exists():
            shutil.rmtree(CHUNK_CACHE)

    if DELETE_ARCHIVED_LEGACY_CACHE and LEGACY_CACHE.exists():
        removed = 0

        for item in completed:
            legacy = LEGACY_CACHE / item['cache_file']

            if (
                legacy.is_file()
                and sha(legacy) == item['cache_sha256']
            ):
                legacy.unlink()
                removed += 1

        print(
            f'Removed {removed:,} safely archived legacy cache PNGs',
            flush=True,
        )

    print(json.dumps(result, indent=2))
    print('Archive:', CHUNK_ARCHIVE)
    print('SHA-256 sidecar:', CHUNK_ARCHIVE_SHA)

    if progress['is_last_chunk']:
        print(
            'This was the final chunk. All eligible candidates have now '
            'been partitioned across chunk indices 0..'
            f'{CHUNK_INDEX}.'
        )
    else:
        print(
            f'Next run: set CHUNK_INDEX = {CHUNK_INDEX + 1}. '
            'You may edit CHUNK_TILE_COUNT for that new chunk; its start '
            'offset will be taken from this chunk\'s recorded selection_stop.'
        )

    return result
