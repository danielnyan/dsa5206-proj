class BudgetStop(Exception):
    pass


def check_deadline(deadline):
    if deadline is not None and time.monotonic() >= deadline:
        raise BudgetStop('Configured session work budget reached')


def sha(path):
    path = Path(path)
    with path.open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()


def atomic_json(path, value):
    path = Path(path)
    temporary = path.with_name(path.name + '.tmp')
    temporary.write_text(json.dumps(value, indent=2, allow_nan=False) + '\n')
    temporary.replace(path)


def append_audit(path, value):
    path = Path(path)
    if path.exists() and path.stat().st_size:
        with path.open('rb+') as stream:
            stream.seek(-1, 2)
            if stream.read(1) != b'\n':
                stream.write(b'\n')
    with path.open('a', encoding='utf-8') as stream:
        stream.write(json.dumps(value, allow_nan=False) + '\n')


def read_jsonl(path):
    path = Path(path)
    if path.exists():
        with path.open() as stream:
            for line in stream:
                try:
                    yield json.loads(line)
                except json.JSONDecodeError:
                    warnings.warn(f'Ignoring interrupted JSONL record in {path.name}')


def versions():
    result = {}
    for name in (
        'numpy', 'Pillow', 'openslide-python', 'openslide-bin',
        'staintools', 'spams-bin', 'opencv-python',
    ):
        try:
            result[name] = importlib.metadata.version(name)
        except importlib.metadata.PackageNotFoundError:
            pass
    return result


def restore_previous(previous, output):
    output = Path(output)
    if previous is not None:
        previous = Path(previous)
        if not previous.is_dir():
            raise FileNotFoundError(f'Previous output does not exist: {previous}')
        if not output.exists():
            size = sum(p.stat().st_size for p in previous.rglob('*') if p.is_file())
            output.parent.mkdir(parents=True, exist_ok=True)
            if shutil.disk_usage(output.parent).free < size + 1024**3:
                raise RuntimeError('Insufficient disk space to restore previous output')
            shutil.copytree(previous, output)
        elif not (output / 'inventory_settings.json').exists():
            raise ValueError('Existing output is not a resumable preparation directory')
    output.mkdir(parents=True, exist_ok=True)
    return output

def discover_data(root):
    root = Path(root)
    table = root / 'train.csv'
    images = root / 'train_images'
    masks = root / 'train_label_masks'

    if not (table.is_file() and images.is_dir() and masks.is_dir()):
        raise FileNotFoundError(
            'DATA_ROOT must contain train.csv, train_images/, and train_label_masks/. '
            'Attach the official PANDA competition data with Add Input.'
        )

    with table.open(newline='') as stream:
        records = list(csv.DictReader(stream))

    required = {'image_id', 'data_provider', 'gleason_score', 'isup_grade'}
    if not records or not required <= records[0].keys():
        raise ValueError('train.csv lacks required PANDA columns')

    ids = [r['image_id'] for r in records]
    if len(ids) != len(set(ids)) or any(not x.isalnum() for x in ids):
        raise ValueError('Invalid or duplicate image IDs')

    usable, excluded = [], []
    for row in records:
        reason = ''
        if row['data_provider'].lower() != 'radboud':
            reason = 'provider_has_no_GP3_GP4_GP5_pixel_labels'
        elif not (images / f"{row['image_id']}.tiff").is_file():
            reason = 'missing_image'
        elif not (masks / f"{row['image_id']}_mask.tiff").is_file():
            reason = 'missing_mask'

        if reason:
            excluded.append({
                'image_id': row['image_id'],
                'data_provider': row['data_provider'],
                'reason': reason,
            })
        else:
            usable.append(row)

    if not usable:
        raise ValueError('No usable Radboud image/mask pairs were found')

    return usable, excluded


def physical_tile_geometry(mpp_x, mpp_y, target_fov_um=150.0):
    """Return native level-0 crop dimensions for a target physical FOV."""
    if not all(math.isfinite(v) and v > 0 for v in (mpp_x, mpp_y)):
        raise ValueError(f'Invalid MPP: {mpp_x}, {mpp_y}')

    width = max(1, int(round(target_fov_um / mpp_x)))
    height = max(1, int(round(target_fov_um / mpp_y)))

    actual_x = width * mpp_x
    actual_y = height * mpp_y

    return width, height, actual_x, actual_y


def mask_region(mask_slide, x, y, width, height):
    rgba = np.asarray(mask_slide.read_region((x, y), 0, (width, height)))
    if rgba.shape != (height, width, 4):
        raise ValueError('Unexpected mask-region shape')
    if not np.all(rgba[:, :, 3] == 255):
        raise ValueError('Mask region contains transparent/out-of-bounds pixels')

    values = rgba[:, :, 0]
    if not np.isin(values, [0, 1, 2, 3, 4, 5]).all():
        raise ValueError('Unexpected Radboud mask encoding')
    return values


def classify_mask(values):
    hist = np.bincount(values.ravel(), minlength=6)
    patterns = [gp for gp in (3, 4, 5) if hist[gp] > 0]

    if len(patterns) == 1:
        status = 'single_gp'
        label = patterns[0] - 3
    elif len(patterns) > 1:
        status = 'mixed'
        label = -1
    else:
        status = 'non_tumour'
        label = -1

    gp_pixels = int(hist[3] + hist[4] + hist[5])
    tumour_fraction = gp_pixels / values.size

    return hist, patterns, status, label, tumour_fraction


def rgb_region(slide, x, y, width, height):
    rgba = slide.read_region((x, y), 0, (width, height))
    background = Image.new('RGBA', rgba.size, (255, 255, 255, 255))
    return Image.alpha_composite(background, rgba).convert('RGB')

class PersistentMaskReadError(RuntimeError):
    """
    Raised only when a PANDA mask cannot be opened/read successfully
    after bounded fresh-handle retries.

    These errors are treated as slide-level source exclusions so that
    inventory may continue. Other unexpected errors remain fatal.
    """

    def __init__(
        self,
        path,
        message,
        x=None,
        y=None,
        width=None,
        height=None,
        attempts=None,
    ):
        super().__init__(message)

        self.path = str(path)
        self.x = x
        self.y = y
        self.width = width
        self.height = height
        self.attempts = attempts


class RetryingMaskSlide:
    """
    OpenSlide-compatible wrapper for PANDA label masks.

    Normal case:
        read_region() behaves exactly like OpenSlide.read_region().

    On OpenSlideError:
        1. close the failed handle;
        2. reopen the same TIFF;
        3. retry the exact same region.

    After RETRIES unsuccessful fresh-handle attempts, raise
    PersistentMaskReadError.

    This wrapper intentionally does NOT fabricate background pixels,
    skip individual mask blocks, or change labels.
    """

    def __init__(
        self,
        path,
        retries=3,
        retry_delay_seconds=1.0,
    ):
        self.path = Path(path)
        self.retries = int(retries)
        self.retry_delay_seconds = float(retry_delay_seconds)
        self._slide = None

        if self.retries < 1:
            raise ValueError('retries must be at least 1')

        self._open_with_retry()

    def _close_current(self):
        if self._slide is not None:
            try:
                self._slide.close()
            except Exception:
                pass

        self._slide = None

    def _open_with_retry(self):
        last_error = None

        for attempt in range(1, self.retries + 1):
            try:
                self._close_current()

                self._slide = openslide.OpenSlide(
                    str(self.path)
                )

                return

            except openslide.OpenSlideError as error:
                last_error = error
                self._close_current()

                print(
                    f'OpenSlide mask open failed: '
                    f'{self.path.name}, '
                    f'attempt={attempt}/{self.retries}: '
                    f'{error}',
                    flush=True,
                )

                if attempt < self.retries:
                    time.sleep(
                        self.retry_delay_seconds
                    )

        raise PersistentMaskReadError(
            path=self.path,
            message=(
                f'Unable to open mask after '
                f'{self.retries} fresh attempts: '
                f'{self.path.name}: {last_error}'
            ),
            attempts=self.retries,
        ) from last_error

    @property
    def dimensions(self):
        return self._slide.dimensions

    @property
    def properties(self):
        return self._slide.properties

    def read_region(
        self,
        location,
        level,
        size,
    ):
        x, y = location
        width, height = size

        last_error = None

        for attempt in range(1, self.retries + 1):
            try:
                return self._slide.read_region(
                    location,
                    level,
                    size,
                )

            except openslide.OpenSlideError as error:
                last_error = error

                print(
                    f'OpenSlide mask read failed: '
                    f'{self.path.name}, '
                    f'x={x}, y={y}, '
                    f'w={width}, h={height}, '
                    f'attempt={attempt}/{self.retries}: '
                    f'{error}',
                    flush=True,
                )

                # OpenSlide errors are latched. Never reuse this handle.
                self._close_current()

                if attempt >= self.retries:
                    break

                time.sleep(
                    self.retry_delay_seconds
                )

                try:
                    self._slide = openslide.OpenSlide(
                        str(self.path)
                    )

                    print(
                        f'Reopened mask successfully for retry: '
                        f'{self.path.name}',
                        flush=True,
                    )

                except openslide.OpenSlideError as reopen_error:
                    last_error = reopen_error

                    print(
                        f'Mask reopen failed: '
                        f'{self.path.name}, '
                        f'attempt={attempt}/{self.retries}: '
                        f'{reopen_error}',
                        flush=True,
                    )

                    self._slide = None

                    # The next loop iteration will need a valid handle.
                    # Try opening it explicitly there.
                    if attempt < self.retries:
                        try:
                            self._slide = openslide.OpenSlide(
                                str(self.path)
                            )
                        except openslide.OpenSlideError:
                            pass

        self._close_current()

        raise PersistentMaskReadError(
            path=self.path,
            message=(
                f'Mask region failed after '
                f'{self.retries} fresh-handle attempts: '
                f'{self.path.name}, '
                f'x={x}, y={y}, '
                f'w={width}, h={height}; '
                f'last error: {last_error}'
            ),
            x=x,
            y=y,
            width=width,
            height=height,
            attempts=self.retries,
        ) from last_error

    def close(self):
        self._close_current()

    def __enter__(self):
        return self

    def __exit__(
        self,
        exc_type,
        exc_value,
        traceback,
    ):
        self.close()
        return False


def init_inventory(
    output,
    signature,
):
    output = Path(output)
    output.mkdir(
        parents=True,
        exist_ok=True,
    )

    marker = (
        output
        / 'inventory_settings.json'
    )

    if (
        marker.exists()
        and json.loads(
            marker.read_text()
        ) != signature
    ):
        raise ValueError(
            'Inventory settings changed. '
            'Use a fresh PREP_ROOT.'
        )

    atomic_json(
        marker,
        signature,
    )

    db = sqlite3.connect(
        output
        / 'inventory.sqlite'
    )

    db.row_factory = sqlite3.Row

    db.executescript(
        '''
        CREATE TABLE IF NOT EXISTS slides (
            image_id TEXT PRIMARY KEY,
            split TEXT,
            role TEXT,
            gleason_score TEXT,
            expected_gp INTEGER,
            patterns TEXT,
            width INTEGER,
            height INTEGER,
            mpp_x REAL,
            mpp_y REAL,
            tile_width INTEGER,
            tile_height INTEGER,
            actual_fov_um_x REAL,
            actual_fov_um_y REAL,
            image_bytes INTEGER,
            mask_bytes INTEGER,
            image_quickhash TEXT,
            mask_quickhash TEXT,
            mask_counts TEXT,
            discarded_edge_pixels INTEGER,
            complete INTEGER
        );

        CREATE TABLE IF NOT EXISTS tiles (
            tile_id TEXT PRIMARY KEY,
            image_id TEXT,
            x INTEGER,
            y INTEGER,
            tile_width INTEGER,
            tile_height INTEGER,
            status TEXT,
            label INTEGER,
            tumour_fraction REAL,
            gp3_pixels INTEGER,
            gp4_pixels INTEGER,
            gp5_pixels INTEGER
        );

        CREATE INDEX IF NOT EXISTS tiles_slide
        ON tiles(image_id);

        CREATE INDEX IF NOT EXISTS tiles_status_label
        ON tiles(status, label);
        '''
    )

    return db


def build_inventory(
    data_root,
    output,
    seed=42,
    target_fov_um=150.0,
    min_mpp=.35,
    max_mpp=.65,
    deadline=None,
):
    data_root = Path(data_root)
    output = Path(output)

    records, excluded = discover_data(
        data_root
    )

    ids = sorted(
        r['image_id']
        for r in records
    )

    shuffled = ids.copy()

    random.Random(
        seed
    ).shuffle(
        shuffled
    )

    heldout = set(
        shuffled[
            :max(
                1,
                round(
                    .2 * len(shuffled)
                ),
            )
        ]
    )

    signature = {
        'pipeline_version': PIPELINE_VERSION,
        'train_csv_sha256': sha(
            data_root
            / 'train.csv'
        ),
        'image_ids': ids,
        'seed': seed,
        'target_fov_um': target_fov_um,
        'output_pixels': OUTPUT_PIXELS,
        'level': 0,
        'min_mpp': min_mpp,
        'max_mpp': max_mpp,
        'split': (
            '80/20 slide/image-id groups; '
            'patient independence unverified'
        ),
        'provider': 'radboud',
    }

    db = init_inventory(
        output,
        signature,
    )

    atomic_json(
        output
        / 'source_exclusions.json',
        excluded,
    )

    atomic_json(
        output
        / 'split.json',
        {
            image_id: (
                'validation'
                if image_id in heldout
                else 'train'
            )
            for image_id in ids
        },
    )

    by_id = {
        r['image_id']: r
        for r in records
    }

    started = time.monotonic()

    for number, image_id in enumerate(
        ids,
        1,
    ):
        image_path = (
            data_root
            / 'train_images'
            / f'{image_id}.tiff'
        )

        mask_path = (
            data_root
            / 'train_label_masks'
            / f'{image_id}_mask.tiff'
        )

        row_csv = by_id[
            image_id
        ]

        split = (
            'validation'
            if image_id in heldout
            else 'train'
        )

        old = db.execute(
            '''
            SELECT *
            FROM slides
            WHERE image_id=?
            ''',
            (image_id,),
        ).fetchone()

        if (
            old
            and old['complete']
        ):
            if (
                old['image_bytes']
                != image_path.stat().st_size
                or old['mask_bytes']
                != mask_path.stat().st_size
            ):
                raise ValueError(
                    f'Source file size changed '
                    f'for {image_id}'
                )

            continue

        # Initialise metadata so that, if mask decoding permanently fails,
        # the slide can still be entered into the inventory as an audited
        # excluded source and the next slide can be processed.
        width = None
        height = None

        mpp_x = None
        mpp_y = None

        tile_w = None
        tile_h = None

        actual_fov_x = None
        actual_fov_y = None

        discarded_edge_pixels = None

        expected_gp = PURE_SCORE_TO_GP.get(
            row_csv['gleason_score']
        )

        image_quickhash = ''
        mask_quickhash = ''

        pixel_counts = np.zeros(
            6,
            dtype=np.int64,
        )

        patterns = set()

        try:
            check_deadline(
                deadline
            )

            # Open the RGB WSI normally.
            #
            # Only PANDA label-mask reads receive the bounded
            # reopen-and-retry behavior.
            with openslide.OpenSlide(
                str(image_path)
            ) as image:

                width, height = (
                    image.dimensions
                )

                mpp_x = float(
                    image.properties.get(
                        'openslide.mpp-x',
                        'nan',
                    )
                )

                mpp_y = float(
                    image.properties.get(
                        'openslide.mpp-y',
                        'nan',
                    )
                )

                image_quickhash = (
                    image.properties.get(
                        'openslide.quickhash-1',
                        '',
                    )
                )

                if not all(
                    math.isfinite(v)
                    and min_mpp <= v <= max_mpp
                    for v in (
                        mpp_x,
                        mpp_y,
                    )
                ):
                    raise ValueError(
                        f'Unverified/out-of-range '
                        f'level-0 MPP: '
                        f'{mpp_x}, {mpp_y}; '
                        f'review scale before '
                        f'changing limits'
                    )

                (
                    tile_w,
                    tile_h,
                    actual_fov_x,
                    actual_fov_y,
                ) = physical_tile_geometry(
                    mpp_x,
                    mpp_y,
                    target_fov_um,
                )

                discarded_edge_pixels = (
                    width * height
                    - (
                        width
                        // tile_w
                        * tile_w
                    )
                    * (
                        height
                        // tile_h
                        * tile_h
                    )
                )

                with RetryingMaskSlide(
                    mask_path,
                    retries=3,
                    retry_delay_seconds=1.0,
                ) as mask:

                    if (
                        image.dimensions
                        != mask.dimensions
                    ):
                        raise ValueError(
                            'Image/mask level-0 '
                            'dimensions differ'
                        )

                    mask_quickhash = (
                        mask.properties.get(
                            'openslide.quickhash-1',
                            '',
                        )
                    )

                    patches = []

                    # Non-overlapping physical-FOV grid.
                    #
                    # Partial edges are still scanned for whole-slide
                    # GP-pattern accounting but are not retained as
                    # training patches.
                    for y in range(
                        0,
                        height,
                        tile_h,
                    ):
                        for x in range(
                            0,
                            width,
                            tile_w,
                        ):
                            check_deadline(
                                deadline
                            )

                            tw = min(
                                tile_w,
                                width - x,
                            )

                            th = min(
                                tile_h,
                                height - y,
                            )

                            (
                                hist,
                                gps,
                                status,
                                label,
                                tumour_fraction,
                            ) = classify_mask(
                                mask_region(
                                    mask,
                                    x,
                                    y,
                                    tw,
                                    th,
                                )
                            )

                            pixel_counts += (
                                hist
                            )

                            patterns.update(
                                gps
                            )

                            if (
                                tw == tile_w
                                and th == tile_h
                            ):
                                patches.append(
                                    (
                                        (
                                            f'{image_id}'
                                            f'_x{x}'
                                            f'_y{y}'
                                        ),
                                        image_id,
                                        x,
                                        y,
                                        tile_w,
                                        tile_h,
                                        status,
                                        label,
                                        float(
                                            tumour_fraction
                                        ),
                                        int(
                                            hist[3]
                                        ),
                                        int(
                                            hist[4]
                                        ),
                                        int(
                                            hist[5]
                                        ),
                                    )
                                )

                    pattern_list = sorted(
                        patterns
                    )

                    if (
                        expected_gp
                        is not None
                        and pattern_list
                        == [expected_gp]
                    ):
                        role = (
                            'pure_score_and_mask_match'
                        )

                    elif (
                        expected_gp
                        is not None
                    ):
                        role = (
                            'pure_score_'
                            'mask_mismatch_or_mixed'
                        )

                    elif pattern_list:
                        role = (
                            'mixed_or_nonpure_score'
                        )

                    else:
                        role = (
                            'no_gp_pixels'
                        )

                    # Nothing is committed until the entire mask has
                    # been scanned successfully. Therefore a later
                    # decoding failure cannot leave partial tiles in
                    # the training inventory.
                    with db:
                        db.execute(
                            '''
                            DELETE FROM tiles
                            WHERE image_id=?
                            ''',
                            (image_id,),
                        )

                        db.executemany(
                            '''
                            INSERT INTO tiles
                            VALUES (
                                ?,?,?,?,?,?,
                                ?,?,?,?,?,?
                            )
                            ''',
                            patches,
                        )

                        db.execute(
                            '''
                            INSERT OR REPLACE
                            INTO slides
                            VALUES (
                                ?,?,?,?,?,?,
                                ?,?,?,?,?,?,
                                ?,?,?,?,?,?,
                                ?,?,?
                            )
                            ''',
                            (
                                image_id,
                                split,
                                role,
                                row_csv[
                                    'gleason_score'
                                ],
                                expected_gp,
                                json.dumps(
                                    pattern_list
                                ),
                                width,
                                height,
                                mpp_x,
                                mpp_y,
                                tile_w,
                                tile_h,
                                actual_fov_x,
                                actual_fov_y,
                                image_path.stat().st_size,
                                mask_path.stat().st_size,
                                image_quickhash,
                                mask_quickhash,
                                json.dumps(
                                    pixel_counts.tolist()
                                ),
                                discarded_edge_pixels,
                                1,
                            ),
                        )

        except PersistentMaskReadError as error:
            # ------------------------------------------------------
            # Persistent mask decoder failure:
            #
            # Exclude the WHOLE slide, audit it, mark it complete in
            # the inventory, and continue.
            #
            # We deliberately do not retain partially scanned tiles:
            # whole-slide GP purity cannot be established reliably.
            # ------------------------------------------------------

            event = {
                'image_id': image_id,
                'split': split,
                'mask_file': (
                    mask_path.name
                ),
                'error': type(
                    error
                ).__name__,
                'message': str(
                    error
                ),
                'x': error.x,
                'y': error.y,
                'width': error.width,
                'height': error.height,
                'attempts': error.attempts,
                'gleason_score': row_csv[
                    'gleason_score'
                ],
                'expected_gp': expected_gp,
                'policy': (
                    'exclude entire slide after '
                    'bounded fresh-handle mask-read '
                    'retries; retain no partial tiles'
                ),
            }

            append_audit(
                output
                / 'mask_read_exclusions.jsonl',
                event,
            )

            warnings.warn(
                (
                    f'Excluded slide {image_id} '
                    f'after persistent mask-read '
                    f'failure: {error}'
                ),
                RuntimeWarning,
            )

            # Mark the slide as processed/excluded so that:
            #
            #   * the overall inventory can complete;
            #   * resumed runs do not retry it forever;
            #   * it contributes zero training/validation tiles.
            #
            # Any partially accumulated mask counts are deliberately
            # NOT treated as valid whole-slide annotation counts.
            with db:
                db.execute(
                    '''
                    DELETE FROM tiles
                    WHERE image_id=?
                    ''',
                    (image_id,),
                )

                db.execute(
                    '''
                    INSERT OR REPLACE
                    INTO slides
                    VALUES (
                        ?,?,?,?,?,?,
                        ?,?,?,?,?,?,
                        ?,?,?,?,?,?,
                        ?,?,?
                    )
                    ''',
                    (
                        image_id,
                        split,
                        'mask_read_failure',
                        row_csv[
                            'gleason_score'
                        ],
                        expected_gp,
                        json.dumps([]),
                        width,
                        height,
                        mpp_x,
                        mpp_y,
                        tile_w,
                        tile_h,
                        actual_fov_x,
                        actual_fov_y,
                        image_path.stat().st_size,
                        mask_path.stat().st_size,
                        image_quickhash,
                        mask_quickhash,
                        # Empty/invalid whole-slide count,
                        # rather than misleading partial counts.
                        json.dumps(
                            [0, 0, 0, 0, 0, 0]
                        ),
                        (
                            discarded_edge_pixels
                            if discarded_edge_pixels
                            is not None
                            else 0
                        ),
                        1,
                    ),
                )

            print(
                f'CPU inventory '
                f'{number}/{len(ids)} slides; '
                f'EXCLUDED {image_id} '
                f'(persistent mask read failure); '
                f'elapsed '
                f'{(time.monotonic() - started) / 60:.1f} min',
                flush=True,
            )

            continue

        except BudgetStop:
            db.close()

            report = inventory_report(
                output,
                min_tumour_fraction=(
                    MIN_TUMOUR_FRACTION
                ),
            )

            report[
                'stop_reason'
            ] = (
                'session_budget; '
                'resume uncommitted slide '
                'next run'
            )

            atomic_json(
                output
                / 'inventory_report.json',
                report,
            )

            return report

        except Exception as error:
            # Unexpected problems remain fail-closed.
            #
            # Examples:
            #   * invalid MPP metadata
            #   * image/mask geometry mismatch
            #   * unexpected mask labels
            #   * SQLite/schema errors
            #
            # Those should not silently remove data.
            append_audit(
                output
                / 'inventory_errors.jsonl',
                {
                    'image_id': (
                        image_id
                    ),
                    'error': type(
                        error
                    ).__name__,
                    'message': str(
                        error
                    ),
                },
            )

            db.close()

            raise

        print(
            f'CPU inventory '
            f'{number}/{len(ids)} slides; '
            f'elapsed '
            f'{(time.monotonic() - started) / 60:.1f} min',
            flush=True,
        )

    db.close()

    report = inventory_report(
        output,
        min_tumour_fraction=(
            MIN_TUMOUR_FRACTION
        ),
    )

    atomic_json(
        output
        / 'inventory_report.json',
        report,
    )

    return report


# %% A3.4 — eligibility, reporting, deterministic selection

def eligibility_clause(min_tumour_fraction=0.0):
    # Training: strict Tolkach-like pure-case proxy: published pure slide score and
    # whole-mask GP pattern agree, plus a single-GP tile.
    # Validation: held-out slide, single-GP tile. No claim of patient independence.
    return (
        "t.status='single_gp' AND t.tumour_fraction>=? AND "
        "(s.split='validation' OR s.role='pure_score_and_mask_match')"
    )


def inventory_report(output, min_tumour_fraction=0.0):
    db = sqlite3.connect(Path(output) / 'inventory.sqlite')
    db.row_factory = sqlite3.Row

    counts = [dict(row) for row in db.execute('''
        SELECT s.split,s.role,t.status,t.label,COUNT(*) AS tiles
        FROM tiles t JOIN slides s USING(image_id)
        GROUP BY s.split,s.role,t.status,t.label
    ''')]

    eligible = [dict(row) for row in db.execute(
        'SELECT s.split,t.label,COUNT(*) AS tiles,COUNT(DISTINCT t.image_id) AS slides '
        'FROM tiles t JOIN slides s USING(image_id) '
        'WHERE ' + eligibility_clause(min_tumour_fraction) + ' '
        'GROUP BY s.split,t.label',
        (min_tumour_fraction,),
    )]

    patterns = [dict(row) for row in db.execute('''
        SELECT split,role,patterns,COUNT(*) AS slides
        FROM slides GROUP BY split,role,patterns
    ''')]

    scale = dict(db.execute('''
        SELECT
          MIN(mpp_x) AS min_mpp_x, MAX(mpp_x) AS max_mpp_x,
          MIN(mpp_y) AS min_mpp_y, MAX(mpp_y) AS max_mpp_y,
          MIN(actual_fov_um_x) AS min_fov_x, MAX(actual_fov_um_x) AS max_fov_x,
          MIN(actual_fov_um_y) AS min_fov_y, MAX(actual_fov_um_y) AS max_fov_y,
          MIN(tile_width) AS min_tile_width, MAX(tile_width) AS max_tile_width,
          MIN(tile_height) AS min_tile_height, MAX(tile_height) AS max_tile_height,
          COUNT(*) AS slides
        FROM slides
    ''').fetchone())

    db.close()

    settings = json.loads((Path(output) / 'inventory_settings.json').read_text())
    expected = len(settings['image_ids'])

    return {
        'complete': scale['slides'] == expected,
        'expected_slides': expected,
        'classes': CLASSES,
        'annotation_counts': counts,
        'eligible_counts': eligible,
        'whole_slide_pattern_counts': patterns,
        'scale': scale,
        'target_fov_um': settings['target_fov_um'],
        'output_pixels': settings['output_pixels'],
        'geometry': 'level-0 MPP-aware ~150um non-overlapping crops; direct LANCZOS resize to 350',
        'training_seed_definition': 'pure 3+3/4+4/5+5 score AND matching single-GP whole-mask proxy; single-GP tile',
        'min_tumour_fraction_for_report': min_tumour_fraction,
        'patient_independence': 'unverified: public image IDs are not patient identifiers',
        'notes': [
            'Karolinska excluded: no GP3/GP4/GP5-resolved pixel labels',
            'tumour_fraction is stored for sensitivity analysis',
            'source optical sampling is not identical to Tolkach even when physical FOV is matched',
        ],
    }


def select_tiles(output, caps, seed=42, min_tumour_fraction=0.0):
    db = sqlite3.connect(Path(output) / 'inventory.sqlite')
    db.row_factory = sqlite3.Row
    selected = []

    for split in ('train', 'validation'):
        for label in range(3):
            limit = caps.get(split)
            if limit is not None:
                limit = int(limit)
                if limit < 1:
                    raise ValueError('Per-class caps must be positive or None')

            query = (
                'SELECT t.*,s.split,s.role,s.gleason_score,s.expected_gp,'
                's.mpp_x,s.mpp_y,s.actual_fov_um_x,s.actual_fov_um_y '
                'FROM tiles t JOIN slides s USING(image_id) '
                'WHERE ' + eligibility_clause(min_tumour_fraction) + ' '
                'AND s.split=? AND t.label=?'
            )

            rows = db.execute(query, (min_tumour_fraction, split, label))

            if limit is None:
                selected.extend(dict(row) for row in rows)
                continue

            heap = []
            for row in rows:
                key = int(hashlib.sha256(f'{seed}:{row["tile_id"]}'.encode()).hexdigest(), 16)
                item = (-key, row['tile_id'], dict(row))
                if len(heap) < limit:
                    heapq.heappush(heap, item)
                elif key < -heap[0][0]:
                    heapq.heapreplace(heap, item)
            selected.extend(item[2] for item in heap)

    db.close()
    return sorted(selected, key=lambda r: (r['image_id'], r['y'], r['x']))


# %% A3.5 — stain normalisation and cache

def make_normalizer(reference):
    import staintools

    normalizer = staintools.StainNormalizer(method='macenko')
    normalizer.fit(staintools.read_image(str(reference)))

    if hasattr(staintools, 'BrightnessStandardizer'):
        brightness = staintools.BrightnessStandardizer().transform
    else:
        brightness = staintools.LuminosityStandardizer.standardize

    return brightness, normalizer


def normalize_tile(tile, normalizers, output_pixels=350):
    brightness, normalizer = normalizers

    # Directly map the ~150 µm source crop to the model's 350x350 input.
    pixels = np.asarray(
        tile.convert('RGB').resize(
            (output_pixels, output_pixels),
            Image.Resampling.LANCZOS,
        )
    )

    output = normalizer.transform(brightness(pixels))

    if (
        output.dtype != np.uint8
        or output.shape != (output_pixels, output_pixels, 3)
        or not np.isfinite(output).all()
    ):
        raise ValueError('Invalid normalized uint8 RGB output')

    return output


def ensure_reference(path):
    path = Path(path)
    if not path.exists():
        path.parent.mkdir(parents=True, exist_ok=True)
        temporary = path.with_suffix('.download')
        with urllib.request.urlopen(REFERENCE_URL, timeout=90) as response, temporary.open('wb') as stream:
            shutil.copyfileobj(response, stream)
        if sha(temporary) != REFERENCE_SHA:
            raise ValueError('Downloaded stain reference checksum mismatch')
        temporary.replace(path)

    if sha(path) != REFERENCE_SHA:
        raise ValueError('Stain reference checksum mismatch')

    return path


def write_manifest(path, rows):
    fields = [
        'tile_id', 'image_id', 'x', 'y',
        'tile_width', 'tile_height',
        'mpp_x', 'mpp_y', 'actual_fov_um_x', 'actual_fov_um_y',
        'split', 'role', 'gleason_score', 'expected_gp',
        'label', 'tumour_fraction',
        'gp3_pixels', 'gp4_pixels', 'gp5_pixels',
        'cache_file', 'cache_sha256',
    ]

    with Path(path).open('w', newline='') as stream:
        writer = csv.DictWriter(stream, fieldnames=fields, extrasaction='ignore')
        writer.writeheader()
        writer.writerows(rows)


def prepare_selected(
    data_root,
    output,
    caps,
    reference,
    min_tumour_fraction=0.0,
    max_cache_gib=8,
    normalizers_override=None,
    deadline=None,
):
    data_root = Path(data_root)
    output = Path(output)

    if not inventory_report(output, min_tumour_fraction)['complete']:
        raise ValueError('Finish the inventory before selecting a subset')

    seed = json.loads((output / 'inventory_settings.json').read_text())['seed']
    selected = select_tiles(
        output,
        caps,
        seed=seed,
        min_tumour_fraction=min_tumour_fraction,
    )

    available_train = {r['label'] for r in selected if r['split'] == 'train'}
    if available_train != {0, 1, 2} or not any(r['split'] == 'validation' for r in selected):
        raise ValueError(
            'Need eligible GP3/GP4/GP5 training seeds and held-out validation. '
            'Inspect inventory_report.json instead of relabelling mixed/Karolinska slides.'
        )

    cache = output / 'cache'
    cache.mkdir(exist_ok=True)

    inventory_settings = json.loads((output / 'inventory_settings.json').read_text())
    settings = {
        'pipeline_version': PIPELINE_VERSION,
        'caps': caps,
        'reference_sha256': sha(reference),
        'inventory_settings_sha256': sha(output / 'inventory_settings.json'),
        'normalization': 'direct resize to 350 + legacy brightness + StainTools Macenko',
        'target_fov_um': inventory_settings['target_fov_um'],
        'output_pixels': inventory_settings['output_pixels'],
        'min_tumour_fraction': min_tumour_fraction,
        'versions': versions(),
    }

    marker = output / 'preparation_settings.json'
    if marker.exists() and json.loads(marker.read_text()) != settings:
        raise ValueError('Preparation settings changed; select a fresh PREP_ROOT')
    atomic_json(marker, settings)

    normalizers = normalizers_override if normalizers_override is not None else make_normalizer(reference)

    ledger_path = output / 'tile_ledger.jsonl'
    previous = {row['tile_id']: row for row in read_jsonl(ledger_path)}
    old_excluded = {row['tile_id']: row for row in read_jsonl(output / 'stain_exclusions.jsonl')}

    completed, excluded = [], []
    total_bytes = sum(p.stat().st_size for p in cache.glob('*.png'))
    started = time.monotonic()
    slide = None
    current = None
    stop_reason = None

    try:
        for number, row in enumerate(selected, 1):
            check_deadline(deadline)

            if row['tile_id'] in old_excluded:
                excluded.append(old_excluded[row['tile_id']])
                continue

            cached = cache / f"{row['tile_id']}.png"
            old = previous.get(row['tile_id'])
            if old and cached.is_file() and sha(cached) == old.get('cache_sha256'):
                completed.append(old)
                continue

            # Conservative guard. PNG compression means actual files are usually smaller
            # than the raw RGB upper bound used here.
            if (
                max_cache_gib <= 0
                or total_bytes + OUTPUT_PIXELS * OUTPUT_PIXELS * 3 > max_cache_gib * 1024**3
                or shutil.disk_usage(output).free < 1024**3
            ):
                raise RuntimeError('Cache size/free-space guard reached; choose a smaller bounded subset')

            if current != row['image_id']:
                if slide:
                    slide.close()
                slide = openslide.OpenSlide(
                    str(data_root / 'train_images' / f"{row['image_id']}.tiff")
                )
                current = row['image_id']

            tile = rgb_region(
                slide,
                int(row['x']),
                int(row['y']),
                int(row['tile_width']),
                int(row['tile_height']),
            )

            try:
                pixels = normalize_tile(tile, normalizers, output_pixels=OUTPUT_PIXELS)
            except Exception as error:
                event = {
                    'tile_id': row['tile_id'],
                    'image_id': row['image_id'],
                    'x': row['x'],
                    'y': row['y'],
                    'split': row['split'],
                    'label': row['label'],
                    'error': type(error).__name__,
                    'message': str(error),
                }
                append_audit(output / 'stain_exclusions.jsonl', event)
                excluded.append(event)
                warnings.warn(
                    f'Excluded {row["tile_id"]}: {type(error).__name__}: {error}',
                    RuntimeWarning,
                )
                continue

            temporary = cached.with_suffix('.tmp')
            Image.fromarray(pixels).save(temporary, format='PNG')
            old_bytes = cached.stat().st_size if cached.exists() else 0
            temporary.replace(cached)
            total_bytes += cached.stat().st_size - old_bytes

            item = dict(row, cache_file=cached.name, cache_sha256=sha(cached))
            append_audit(ledger_path, item)
            completed.append(item)

            if number % 100 == 0:
                print(
                    f'CPU cache {number}/{len(selected)} tiles; '
                    f'{total_bytes / 1024**3:.2f} GiB',
                    flush=True,
                )

    except BudgetStop:
        stop_reason = 'session_budget; resume cached tiles from ledger'
    finally:
        if slide:
            slide.close()
        atomic_json(
            output / 'preparation_progress.json',
            {
                'selected': len(selected),
                'retained': len(completed),
                'excluded': len(excluded),
                'cache_bytes': total_bytes,
                'elapsed_seconds': time.monotonic() - started,
            },
        )

    if stop_reason:
        report = {
            'complete': False,
            'training_ready': False,
            'stop_reason': stop_reason,
            'selected': len(selected),
            'retained': len(completed),
            'excluded': len(excluded),
            'settings': settings,
        }
        atomic_json(output / 'prepared.json', report)
        return report

    manifests = output / 'manifests'
    manifests.mkdir(exist_ok=True)

    counts = {}
    for split in ('train', 'validation'):
        rows = [r for r in completed if r['split'] == split]
        write_manifest(manifests / f'{split}.csv', rows)
        counts[split] = {
            CLASSES[k]: sum(r['label'] == k for r in rows)
            for k in range(3)
        }

    ready = all(counts['train'][c] > 0 for c in CLASSES) and sum(counts['validation'].values()) > 0

    atomic_json(
        output / 'stain_exclusions.json',
        {
            'count': len(excluded),
            'tiles': excluded,
            'log': 'stain_exclusions.jsonl',
            'policy': 'exclude tile preprocessing exceptions; other operations abort',
        },
    )

    report = {
        'complete': True,
        'training_ready': ready,
        'counts': counts,
        'selected': len(selected),
        'retained': len(completed),
        'sample_only': any(v is not None for v in caps.values()),
        'sampling': 'deterministic per-class cap; no automatic replacement after exclusions',
        'settings': settings,
        'versions': versions(),
        'target_fov_um': TARGET_FOV_UM,
        'output_pixels': OUTPUT_PIXELS,
        'cache_bytes': total_bytes,
        'patient_independence': 'unverified',
        'elapsed_seconds': time.monotonic() - started,
        'manifests': {
            name: sha(manifests / name)
            for name in ('train.csv', 'validation.csv')
        },
        'stain_exclusion_count': len(excluded),
    }

    report['audit_hashes'] = {
        name: sha(output / name)
        for name in ('split.json', 'inventory_settings.json', 'stain_exclusions.json')
    }

    atomic_json(output / 'prepared.json', report)

    if not ready:
        raise ValueError('After exclusions, preparation lacks training-class coverage or validation')

    return report


# %% A3.6 — load/verify prepared output

def load_prepared(root):
    root = Path(root)
    report = json.loads((root / 'prepared.json').read_text())

    if not report.get('complete') or not report.get('training_ready'):
        raise ValueError('Prepared input is incomplete/not training-ready')

    if abs(float(report['target_fov_um']) - TARGET_FOV_UM) > 1e-9:
        raise ValueError('Unexpected target physical FOV')
    if int(report['output_pixels']) != OUTPUT_PIXELS:
        raise ValueError('Unexpected model input size')

    records = {}
    for split in ('train', 'validation'):
        path = root / 'manifests' / f'{split}.csv'
        if sha(path) != report['manifests'][path.name]:
            raise ValueError('Manifest hash changed')

        with path.open(newline='') as stream:
            records[split] = list(csv.DictReader(stream))

        for row in records[split]:
            if Path(row['cache_file']).name != row['cache_file']:
                raise ValueError('Unsafe cache file name')
            if sha(root / 'cache' / row['cache_file']) != row['cache_sha256']:
                raise ValueError(f'Cache checksum mismatch: {row["tile_id"]}')
            row['label'] = int(row['label'])

    train_ids = {r['image_id'] for r in records['train']}
    val_ids = {r['image_id'] for r in records['validation']}
    if train_ids & val_ids:
        raise ValueError('Slide leakage across split')

    if {r['label'] for r in records['train']} != {0, 1, 2} or not records['validation']:
        raise ValueError('Missing training classes or validation')

    return report, records
