class BudgetStop(Exception):
    """Internal cooperative stop; save outputs and return normally to Kaggle."""


def check_deadline(deadline):
    if deadline is not None and time.monotonic() >= deadline:
        raise BudgetStop('Configured session work budget reached')


def restore_previous(previous, output):
    """Copy a saved notebook output to writable working storage exactly once."""
    output = Path(output)
    if previous is not None:
        previous = Path(previous)
        if not previous.is_dir():
            raise FileNotFoundError(f'Previous output does not exist: {previous}')
        if not output.exists():
            size = sum(p.stat().st_size for p in previous.rglob('*') if p.is_file())
            output.parent.mkdir(parents=True, exist_ok=True)
            if shutil.disk_usage(output.parent).free < size + 1024**3:
                raise RuntimeError('Insufficient disk space to restore previous output')
            shutil.copytree(previous, output)
        elif not (output / 'inventory_settings.json').exists():
            raise ValueError('Existing output is not a resumable preparation directory')
    output.mkdir(parents=True, exist_ok=True)
    return output


def sha(path):
    with Path(path).open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()


def atomic_json(path, value):
    path = Path(path)
    temporary = path.with_name(path.name + '.tmp')
    temporary.write_text(json.dumps(value, indent=2, allow_nan=False) + '\n')
    temporary.replace(path)


def append_audit(path, value):
    path = Path(path)
    if path.exists() and path.stat().st_size:
        with path.open('rb+') as stream:
            stream.seek(-1, 2)
            if stream.read(1) != b'\n':
                stream.write(b'\n')  # Separate an interrupted final JSONL record.
    with path.open('a', encoding='utf-8') as stream:
        stream.write(json.dumps(value, allow_nan=False) + '\n')


def read_jsonl(path):
    path = Path(path)
    if path.exists():
        with path.open() as stream:
            for line in stream:
                try:
                    yield json.loads(line)
                except json.JSONDecodeError:
                    warnings.warn(f'Ignoring interrupted JSONL record in {path.name}')


def versions():
    result = {}
    for name in ('numpy', 'Pillow', 'openslide-python', 'openslide-bin', 'staintools',
                 'spams-bin', 'opencv-python-headless', 'tensorflow', 'keras'):
        try:
            result[name] = importlib.metadata.version(name)
        except importlib.metadata.PackageNotFoundError:
            pass
    return result


def discover_data(root):
    root = Path(root)
    table = root / 'train.csv'
    images = root / 'train_images'
    masks = root / 'train_label_masks'
    if not (table.is_file() and images.is_dir() and masks.is_dir()):
        raise FileNotFoundError('DATA_ROOT must contain train.csv, train_images/, and train_label_masks/. '
                                'Attach the official competition data through Add Input; do not copy all TIFFs.')
    with table.open(newline='') as stream:
        records = list(csv.DictReader(stream))
    if not records or not {'image_id', 'data_provider', 'gleason_score', 'isup_grade'} <= records[0].keys():
        raise ValueError('train.csv lacks required PANDA columns')
    ids = [r['image_id'] for r in records]
    if len(ids) != len(set(ids)) or any(not name.isalnum() for name in ids):
        raise ValueError('Invalid or duplicate image IDs')
    usable, audit = [], []
    for row in records:
        reason = ''
        if row['data_provider'].lower() != 'radboud':
            reason = 'provider_has_no_GP3_GP4_GP5_pixel_labels'
        elif not (images / (row['image_id'] + '.tiff')).is_file():
            reason = 'missing_image'
        elif not (masks / (row['image_id'] + '_mask.tiff')).is_file():
            reason = 'missing_mask'
        if reason:
            audit.append(dict(image_id=row['image_id'], data_provider=row['data_provider'], reason=reason))
        else:
            usable.append(row)
    if not usable:
        raise ValueError('No Radboud image/mask pairs available. The two uploaded sample files alone cannot train GP classes.')
    return usable, audit


def mask_region(mask_slide, x, y, width, height):
    # PANDA mask labels are in the RED channel; RGB channels are not identical.
    rgba = np.asarray(mask_slide.read_region((x, y), 0, (width, height)))
    if not np.all(rgba[:, :, 3] == 255):
        raise ValueError('Mask region contains transparent/out-of-bounds pixels')
    values = rgba[:, :, 0]
    if not np.isin(values, [0, 1, 2, 3, 4, 5]).all():
        raise ValueError('Unexpected Radboud mask encoding')
    return values


def classify_mask(values):
    hist = np.bincount(values.ravel(), minlength=6)
    patterns = [gp for gp in (3, 4, 5) if hist[gp]]
    status = 'single_gp' if len(patterns) == 1 else 'mixed' if patterns else 'non_tumour'
    label = patterns[0] - 3 if status == 'single_gp' else -1
    return hist, patterns, status, label


def rgb_region(slide, x, y):
    rgba = slide.read_region((x, y), 0, (600, 600))
    background = Image.new('RGBA', rgba.size, (255, 255, 255, 255))
    return Image.alpha_composite(background, rgba).convert('RGB')

def init_inventory(output, signature):
    output = Path(output)
    output.mkdir(parents=True, exist_ok=True)
    marker = output / 'inventory_settings.json'
    if marker.exists() and json.loads(marker.read_text()) != signature:
        raise ValueError('Inventory settings changed. Choose a new PREP_ROOT.')
    atomic_json(marker, signature)
    db = sqlite3.connect(output / 'inventory.sqlite')
    db.row_factory = sqlite3.Row
    db.executescript('''
    CREATE TABLE IF NOT EXISTS slides (
        image_id TEXT PRIMARY KEY, split TEXT, role TEXT, patterns TEXT,
        width INTEGER, height INTEGER, mpp_x REAL, mpp_y REAL,
        image_bytes INTEGER, mask_bytes INTEGER, image_quickhash TEXT, mask_quickhash TEXT,
        mask_counts TEXT, discarded_edge_pixels INTEGER, complete INTEGER);
    CREATE TABLE IF NOT EXISTS tiles (
        tile_id TEXT PRIMARY KEY, image_id TEXT, x INTEGER, y INTEGER,
        status TEXT, label INTEGER, gp3_pixels INTEGER, gp4_pixels INTEGER, gp5_pixels INTEGER);
    CREATE INDEX IF NOT EXISTS tiles_slide ON tiles(image_id);
    ''')
    return db


def build_inventory(data_root, output, seed=42, min_mpp=.35, max_mpp=.65, deadline=None):
    data_root, output = Path(data_root), Path(output)
    records, excluded = discover_data(data_root)
    ids = sorted(r['image_id'] for r in records)
    shuffled = ids.copy()
    random.Random(seed).shuffle(shuffled)
    heldout = set(shuffled[:max(1, round(.2 * len(shuffled)))])
    signature = dict(pipeline_version=PIPELINE_VERSION, train_csv_sha256=sha(data_root / 'train.csv'), image_ids=ids,
                     seed=seed, native_tile=600, output_pixels=350, level=0,
                     min_mpp=min_mpp, max_mpp=max_mpp,
                     split='80/20 image-id groups; patient independence unverified', provider='radboud')
    db = init_inventory(output, signature)
    atomic_json(output / 'source_exclusions.json', excluded)
    atomic_json(output / 'split.json', {i: 'validation' if i in heldout else 'train' for i in ids})
    started = time.monotonic()
    for number, image_id in enumerate(ids, 1):
        image_path = data_root / 'train_images' / (image_id + '.tiff')
        mask_path = data_root / 'train_label_masks' / (image_id + '_mask.tiff')
        old = db.execute('SELECT * FROM slides WHERE image_id=?', (image_id,)).fetchone()
        if old and old['complete']:
            if old['image_bytes'] != image_path.stat().st_size or old['mask_bytes'] != mask_path.stat().st_size:
                raise ValueError(f'Source file size changed for {image_id}')
            continue
        try:
            check_deadline(deadline)
            with openslide.OpenSlide(str(image_path)) as image, openslide.OpenSlide(str(mask_path)) as mask:
                if image.dimensions != mask.dimensions:
                    raise ValueError('Image/mask level-0 dimensions differ')
                width, height = image.dimensions
                mpp_x = float(image.properties.get('openslide.mpp-x', 'nan'))
                mpp_y = float(image.properties.get('openslide.mpp-y', 'nan'))
                if not all(math.isfinite(v) and min_mpp <= v <= max_mpp for v in (mpp_x, mpp_y)):
                    raise ValueError(f'Unverified/out-of-range level-0 MPP: {mpp_x}, {mpp_y}; review scale before changing limits')
                pixel_counts = np.zeros(6, dtype=np.int64)
                patches = []
                patterns = set()
                # Read only one 600px mask block at a time, including partial edges.
                # No WSI RGB array is ever loaded into memory.
                for y in range(0, height, 600):
                    for x in range(0, width, 600):
                        check_deadline(deadline)
                        tw, th = min(600, width-x), min(600, height-y)
                        hist, gps, status, label = classify_mask(mask_region(mask, x, y, tw, th))
                        pixel_counts += hist
                        patterns.update(gps)
                        if tw == th == 600:
                            patches.append((f'{image_id}_x{x}_y{y}', image_id, x, y, status,
                                            label, int(hist[3]), int(hist[4]), int(hist[5])))
                role = 'pure' if len(patterns) == 1 else 'mixed' if patterns else 'unknown'
                with db:
                    db.execute('DELETE FROM tiles WHERE image_id=?', (image_id,))
                    db.executemany('INSERT INTO tiles VALUES (?,?,?,?,?,?,?,?,?)', patches)
                    db.execute('INSERT OR REPLACE INTO slides VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)',
                               (image_id, 'validation' if image_id in heldout else 'train', role,
                                json.dumps(sorted(patterns)), width, height, mpp_x, mpp_y,
                                image_path.stat().st_size, mask_path.stat().st_size,
                                image.properties.get('openslide.quickhash-1', ''),
                                mask.properties.get('openslide.quickhash-1', ''),
                                json.dumps(pixel_counts.tolist()), width*height-(width//600*600)*(height//600*600), 1))
        except BudgetStop:
            db.close()
            report = inventory_report(output)
            report['stop_reason'] = 'session_budget; resume uncommitted slide next run'
            atomic_json(output / 'inventory_report.json', report)
            return report
        except Exception as error:
            append_audit(output / 'inventory_errors.jsonl', dict(image_id=image_id, error=type(error).__name__, message=str(error)))
            db.close()
            raise  # Provider/geometry/scale/reader problems must not silently change the cohort.
        print(f'CPU inventory {number}/{len(ids)} slides; elapsed {(time.monotonic()-started)/60:.1f} min', flush=True)
    db.close()
    report = inventory_report(output)
    atomic_json(output / 'inventory_report.json', report)
    return report

def eligible_sql():
    # Same policy as Gleason2019: train seeds from whole-slide single-pattern proxy;
    # validation single-GP tiles may come from any held-out slide role.
    return "t.status='single_gp' AND (s.split='validation' OR s.role='pure')"


def inventory_report(output):
    db = sqlite3.connect(Path(output) / 'inventory.sqlite')
    db.row_factory = sqlite3.Row
    counts = [dict(row) for row in db.execute('''SELECT s.split,s.role,t.status,t.label,COUNT(*) AS tiles
                FROM tiles t JOIN slides s USING(image_id) GROUP BY s.split,s.role,t.status,t.label''')]
    eligible = [dict(row) for row in db.execute('SELECT s.split,t.label,COUNT(*) AS tiles,COUNT(DISTINCT t.image_id) AS slides '
                'FROM tiles t JOIN slides s USING(image_id) WHERE '+eligible_sql()+' GROUP BY s.split,t.label')]
    patterns = [dict(row) for row in db.execute('SELECT split,role,patterns,COUNT(*) AS slides FROM slides GROUP BY split,role,patterns')]
    scale = dict(db.execute('SELECT MIN(mpp_x) AS min_mpp,MAX(mpp_x) AS max_mpp,COUNT(*) AS slides FROM slides').fetchone())
    db.close()
    expected = len(json.loads((Path(output)/'inventory_settings.json').read_text())['image_ids'])
    return dict(complete=scale['slides']==expected, expected_slides=expected, classes=CLASSES, annotation_counts=counts, eligible_counts=eligible,
                whole_slide_pattern_counts=patterns, scale=scale,
                geometry='level-0 native 600x600, stride 600, discard partial edges; RGB LANCZOS to 350',
                seed_definition='single-pattern whole-slide mask proxy, not proven pure patient cases',
                patient_independence='unverified: public image IDs are not patient identifiers',
                notes=['Karolinska excluded: no GP-resolved pixel labels', 'Any resolved GP pixel contributes; no purity percentage invented'])


def select_tiles(output, caps, seed=42):
    db = sqlite3.connect(Path(output) / 'inventory.sqlite')
    db.row_factory = sqlite3.Row
    selected = []
    for split in ('train', 'validation'):
        for label in range(3):
            limit = int(caps[split])
            if limit < 1:
                raise ValueError('Per-class caps must be positive')
            heap = []
            query = ('SELECT t.*,s.split,s.role,s.mpp_x,s.mpp_y FROM tiles t JOIN slides s USING(image_id) '
                     'WHERE '+eligible_sql()+' AND s.split=? AND t.label=?')
            for row in db.execute(query, (split, label)):
                key = int(hashlib.sha256(f'{seed}:{row["tile_id"]}'.encode()).hexdigest(), 16)
                item = (-key, row['tile_id'], dict(row))
                if len(heap) < limit:
                    heapq.heappush(heap, item)
                elif key < -heap[0][0]:
                    heapq.heapreplace(heap, item)
            selected.extend(item[2] for item in heap)
    db.close()
    return sorted(selected, key=lambda r: (r['image_id'], r['y'], r['x']))

def make_normalizer(reference):
    import staintools
    normalizer = staintools.StainNormalizer(method='macenko')
    normalizer.fit(staintools.read_image(str(reference)))  # Raw reference, as in the legacy pipeline.
    if hasattr(staintools, 'BrightnessStandardizer'):
        brightness = staintools.BrightnessStandardizer().transform
    else:
        brightness = staintools.LuminosityStandardizer.standardize
    return brightness, normalizer


def normalize_tile(tile, normalizers):
    brightness, normalizer = normalizers
    pixels = np.asarray(tile.convert('RGB').resize((350, 350), Image.Resampling.LANCZOS))
    output = normalizer.transform(brightness(pixels))
    if output.dtype != np.uint8 or output.shape != (350, 350, 3) or not np.isfinite(output).all():
        raise ValueError('Invalid normalized uint8 RGB output')
    return output


def ensure_reference(path):
    path = Path(path)
    if not path.exists():
        path.parent.mkdir(parents=True, exist_ok=True)
        temporary = path.with_suffix('.download')
        with urllib.request.urlopen(REFERENCE_URL, timeout=90) as response, temporary.open('wb') as stream:
            shutil.copyfileobj(response, stream)
        if sha(temporary) != REFERENCE_SHA:
            raise ValueError('Downloaded stain reference checksum mismatch')
        temporary.replace(path)
    if sha(path) != REFERENCE_SHA:
        raise ValueError('Stain reference checksum mismatch')
    return path


def write_manifest(path, rows):
    fields = ['tile_id','image_id','x','y','split','role','label','gp3_pixels','gp4_pixels','gp5_pixels',
              'mpp_x','mpp_y','cache_file','cache_sha256']
    with Path(path).open('w', newline='') as stream:
        writer = csv.DictWriter(stream, fieldnames=fields, extrasaction='ignore')
        writer.writeheader()
        writer.writerows(rows)

def prepare_selected(data_root, output, caps, reference, max_cache_gib=8, normalizers_override=None, deadline=None):
    data_root, output = Path(data_root), Path(output)
    if not inventory_report(output)['complete']:
        raise ValueError('Finish the inventory before selecting a subset')
    seed = json.loads((output/'inventory_settings.json').read_text())['seed']
    selected = select_tiles(output, caps, seed=seed)
    available_train = {r['label'] for r in selected if r['split']=='train'}
    if available_train != {0, 1, 2} or not any(r['split']=='validation' for r in selected):
        raise ValueError('Need eligible GP3/GP4/GP5 training seeds and held-out validation. Inspect inventory_report.json; '
                         'do not relabel benign/Karolinska/mixed slides or search splits silently.')
    cache = output / 'cache'
    cache.mkdir(exist_ok=True)
    settings = dict(pipeline_version=PIPELINE_VERSION, caps=caps, reference_sha256=sha(reference), inventory_settings_sha256=sha(output/'inventory_settings.json'),
                    normalization='legacy brightness + StainTools 2.1.2 Macenko', output_pixels=350,
                    versions=versions())
    marker = output / 'preparation_settings.json'
    if marker.exists() and json.loads(marker.read_text()) != settings:
        raise ValueError('Preparation settings changed; select a fresh PREP_ROOT')
    atomic_json(marker, settings)
    # Reference fitting remains fatal. Never fall back to another normalization.
    normalizers = normalizers_override if normalizers_override is not None else make_normalizer(reference)
    # Resume completed tile records only; incomplete PNG writes are safely redone.
    ledger_path = output / 'tile_ledger.jsonl'
    previous = {row['tile_id']:row for row in read_jsonl(ledger_path)}
    old_excluded = {row['tile_id']:row for row in read_jsonl(output/'stain_exclusions.jsonl')}
    completed, excluded = [], []
    total_bytes = sum(p.stat().st_size for p in cache.glob('*.png'))
    started = time.monotonic()
    slide = None
    current = None
    stop_reason = None
    try:
        for number, row in enumerate(selected, 1):
            check_deadline(deadline)
            if row['tile_id'] in old_excluded:
                excluded.append(old_excluded[row['tile_id']])
                continue
            cached = cache / (row['tile_id'] + '.png')
            old = previous.get(row['tile_id'])
            if old and cached.is_file() and sha(cached)==old.get('cache_sha256'):
                completed.append(old)
                continue
            if max_cache_gib <= 0 or total_bytes + 350*350*3 > max_cache_gib*1024**3 or shutil.disk_usage(output).free < 1024**3:
                raise RuntimeError('Cache size/free-space guard reached. Save audit and choose a smaller bounded subset.')
            if current != row['image_id']:
                if slide:
                    slide.close()
                slide = openslide.OpenSlide(str(data_root/'train_images'/(row['image_id']+'.tiff')))
                current = row['image_id']
            tile = rgb_region(slide, row['x'], row['y'])
            try:
                pixels = normalize_tile(tile, normalizers)
            except Exception as error:
                event = dict(tile_id=row['tile_id'], image_id=row['image_id'], x=row['x'], y=row['y'],
                             split=row['split'], label=row['label'], error=type(error).__name__, message=str(error))
                append_audit(output/'stain_exclusions.jsonl', event)
                excluded.append(event)
                warnings.warn(f'Excluded {row["tile_id"]}: {type(error).__name__}: {error}', RuntimeWarning)
                continue
            temporary = cached.with_suffix('.tmp')
            Image.fromarray(pixels).save(temporary, format='PNG')
            old_bytes = cached.stat().st_size if cached.exists() else 0
            temporary.replace(cached)
            total_bytes += cached.stat().st_size-old_bytes
            item = dict(row, cache_file=cached.name, cache_sha256=sha(cached))
            append_audit(ledger_path, item)
            completed.append(item)
            if number % 100 == 0:
                print(f'CPU cache {number}/{len(selected)} tiles; {total_bytes/1024**3:.2f} GiB', flush=True)
    except BudgetStop:
        stop_reason = 'session_budget; resume cached tiles from ledger'
    finally:
        if slide:
            slide.close()
        atomic_json(output/'preparation_progress.json', dict(selected=len(selected), retained=len(completed),
                    excluded=len(excluded), cache_bytes=total_bytes, elapsed_seconds=time.monotonic()-started))
    if stop_reason:
        report = dict(complete=False, training_ready=False, stop_reason=stop_reason,
                      selected=len(selected), retained=len(completed), excluded=len(excluded), settings=settings)
        atomic_json(output/'prepared.json', report)
        return report
    manifests = output/'manifests'
    manifests.mkdir(exist_ok=True)
    counts = {}
    for split in ('train','validation'):
        rows = [r for r in completed if r['split']==split]
        write_manifest(manifests/(split+'.csv'), rows)
        counts[split] = {CLASSES[k]:sum(r['label']==k for r in rows) for k in range(3)}
    ready = all(counts['train'][c]>0 for c in CLASSES) and sum(counts['validation'].values())>0
    atomic_json(output/'stain_exclusions.json', dict(count=len(excluded), tiles=excluded,
                log='stain_exclusions.jsonl', policy='exclude tile preprocessing exceptions; other operations abort'))
    report = dict(complete=True, training_ready=ready, counts=counts, selected=len(selected), retained=len(completed),
                  sample_only=True, sampling='deterministic per-class capped pilot; no automatic replacement after exclusions',
                  settings=settings, versions=versions(), native_tile=600, output_pixels=350,
                  cache_bytes=total_bytes, patient_independence='unverified',
                  elapsed_seconds=time.monotonic()-started,
                  manifests={name:sha(manifests/name) for name in ('train.csv','validation.csv')})
    report['stain_exclusion_count'] = len(excluded)
    report['audit_hashes'] = {name:sha(output/name) for name in
                            ('split.json','inventory_settings.json','stain_exclusions.json')}
    atomic_json(output/'prepared.json',report)
    if not ready:
        raise ValueError('After exclusions, preparation lacks seed-class coverage or validation. Inspect audits.')
    return report


def load_prepared(root):
    root = Path(root)
    report = json.loads((root/'prepared.json').read_text())
    if not report.get('complete') or not report.get('training_ready'):
        raise ValueError('Prepared input is incomplete/not training-ready')
    records = {}
    for split in ('train','validation'):
        path=root/'manifests'/(split+'.csv')
        if sha(path)!=report['manifests'][path.name]:
            raise ValueError('Manifest hash changed')
        with path.open(newline='') as stream:
            records[split]=list(csv.DictReader(stream))
        for row in records[split]:
            if Path(row['cache_file']).name!=row['cache_file']:
                raise ValueError('Unsafe cache file name')
            if sha(root/'cache'/row['cache_file'])!=row['cache_sha256']:
                raise ValueError(f'Cache checksum mismatch: {row["tile_id"]}')
            row['label']=int(row['label'])
    if set(r['image_id'] for r in records['train']) & set(r['image_id'] for r in records['validation']):
        raise ValueError('Slide leakage across split')
    if {r['label'] for r in records['train']}!={0,1,2} or not records['validation']:
        raise ValueError('Missing training classes or validation')
    return report,records
